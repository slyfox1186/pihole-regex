# Python packages for scripts/pihole_admin.py
# ===========================================
#
# Only scripts/pihole_admin.py uses these packages. The other scripts
# (pi.py, adlist.py, unbound.py and lint_domain_lists.py) need nothing
# beyond Python itself.
#
# Install everything at once: `pip install -r requirements.txt`
#
# Or install only what you need. Each package loads only when a command uses
# it, so --help, --stats, --backup and --optimize run with none installed.


# Needed for any command that prints a table
# (--report, --top-domains, --query, --advanced-query)
tabulate

# Charts for --report and --analyze-query-trends
matplotlib

# Query-trend analysis for --analyze-query-trends
pandas

# Nicer grid styling on the charts (optional; charts work without it)
seaborn

# Faster --find-similar (without it the script uses fuzzywuzzy if installed,
# otherwise Python's own difflib)
rapidfuzz

# More accurate --categorize-domains (without it the script uses a simpler
# last-two-labels guess)
tld
