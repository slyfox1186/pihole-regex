# Python packages for scripts/pihole_admin.py
# ===========================================
#
# Only scripts/pihole_admin.py requires these packages to work.
#
# All other scripts need nothing more beyond Python itself.
#
# To install the pip packages, execute: pip install -r requirements.txt
#

# Needed for any command that prints a table (--report, --top-domains,
# --query, --advanced-query)
tabulate

# Charts for --report and --analyze-query-trends
matplotlib

# Query-trend analysis for --analyze-query-trends
pandas

# Nicer grid styling on the charts (optional; charts work without it)
seaborn

# Faster --find-similar (without it the script uses fuzzywuzzy if
# installed, otherwise Python's own difflib)
rapidfuzz

# More accurate --categorize-domains (without it the script
# uses a simpler last-two-labels guess)
tld
